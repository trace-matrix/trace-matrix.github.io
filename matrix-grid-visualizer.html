C++20 · O(R·C) Build & O(1) Submatrix Query
struct PrefixSum2D {
    int n, m;
    vector<vector<long long>> p;

    PrefixSum2D(const vector<vector<long long>>& a)
        : n(a.size()), m(a[0].size()), p(n + 1, vector<long long>(m + 1, 0)) {
        for (int r = 0; r < n; ++r) {
            for (int c = 0; c < m; ++c) {
                p[r + 1][c + 1] = a[r][c] + p[r][c + 1] + p[r + 1][c] - p[r][c];
            }
        }
    }

    // Inclusive query for 0-indexed [r1..r2] x [c1..c2]
    long long query(int r1, int c1, int r2, int c2) const {
        return p[r2 + 1][c2 + 1] - p[r1][c2 + 1] - p[r2 + 1][c1] + p[r1][c1];
    }
};
