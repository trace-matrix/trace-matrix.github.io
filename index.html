2D Prefix Sum Struct (1-indexed)

template <typename T = long long>
struct PrefixSum2D {
    int n, m;
    vector<vector<T>> pref;
    PrefixSum2D(int n, int m) : n(n), m(m), pref(n + 1, vector<T>(m + 1, 0)) {}
    
    // Build prefix sum from 1-indexed or 0-indexed matrix
    void build(const vector<vector<T>>& a) {
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                pref[i][j] = a[i - 1][j - 1] + pref[i - 1][j] + pref[i][j - 1] - pref[i - 1][j - 1];
            }
        }
    }
    
    // Query sum in submatrix [r1, c1] to [r2, c2] (1-indexed, inclusive)
    T query(int r1, int c1, int r2, int c2) const {
        if (r1 > r2 || c1 > c2) return 0;
        return pref[r2][c2] - pref[r1 - 1][c2] - pref[r2][c1 - 1] + pref[r1 - 1][c1 - 1];
    }
};

Point Struct & Convex Hull (Andrew's Monotone Chain)

using ll = long long;
struct Point {
    ll x, y;
    bool operator<(const Point& o) const {
        return x != o.x ? x < o.x : y < o.y;
    }
};

// Cross product (B - A) x (C - A)
// > 0: CCW (left turn), < 0: CW (right turn), == 0: Collinear
ll cross(Point A, Point B, Point C) {
    return (B.x - A.x) * (C.y - A.y) - (B.y - A.y) * (C.x - A.x);
}

vector<Point> convex_hull(vector<Point> pts) {
    int n = pts.size(), k = 0;
    if (n <= 2) return pts;
    vector<Point> h(2 * n);
    sort(pts.begin(), pts.end());
    
    // Lower hull
    for (int i = 0; i < n; i++) {
        while (k >= 2 && cross(h[k - 2], h[k - 1], pts[i]) <= 0) k--;
        h[k++] = pts[i];
    }
    // Upper hull
    for (int i = n - 2, t = k + 1; i >= 0; i--) {
        while (k >= t && cross(h[k - 2], h[k - 1], pts[i]) <= 0) k--;
        h[k++] = pts[i];
    }
    h.resize(k - 1);
    return h;
}

Shoelace Formula & Pick's Theorem

// Double signed area (2 * Area) of a polygon
ll double_signed_area(const vector<Point>& p) {
    ll area = 0;
    int n = p.size();
    for (int i = 0; i < n; i++) {
        int j = (i + 1) % n;
        area += p[i].x * p[j].y - p[j].x * p[i].y;
    }
    return area; // Positive: CCW, Negative: CW
}

// Pick's Theorem: Area = I + B / 2 - 1  =>  I = Area - B / 2 + 1
// Where I = strictly interior lattice points, B = boundary lattice points
ll boundary_lattice_points(const vector<Point>& p) {
    ll b = 0;
    int n = p.size();
    for (int i = 0; i < n; i++) {
        int j = (i + 1) % n;
        b += std::gcd(std::abs(p[i].x - p[j].x), std::abs(p[i].y - p[j].y));
    }
    return b;
}
